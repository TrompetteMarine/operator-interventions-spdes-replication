% Engine: finite-rank structure. The first-order studentised field is globally
% bounded (a 2-d loading against one Gaussian vector), so derivative zeros are
% dangerous only through the second-order remainder — which is exactly what
% the floor dominates under the stated rate conditions.
\closuredraft
\begin{proof}
Let $r$ be the family rate ($\bn$ fast, $\an$ slow). By the uniform
second-order expansion of Part S4, uniformly in $\tau\in\cT$,
\begin{equation}\label{eq:stb-exp}
 r\big(\widehat T(\tau)-T_0(\tau)\big)
 =d(\tau)^\top H_{N,n}
 +R_{N,n}(\tau),
\end{equation}
where $H_{N,n}:=\An(\widehat\vartheta-\vzero)=O_{\Pp}(1)$ is the
\emph{common} two-dimensional estimation error, $d(\tau)$ is the
family-specific loading with $s(\tau)^2=d(\tau)^\top J^{-1}d(\tau)$
(for a slow family, $d$ has a zero fast coordinate and
$R_{N,n}$ contains the omitted fast linear term), and
\begin{equation}\label{eq:stb-rem}
 \sup_\tau|R_{N,n}(\tau)|
 =\begin{cases}
 O_{\Pp}(\bn^{-1}), & \text{fast family},\\[1pt]
 O_{\Pp}(\an^{-1})+O_{\Pp}(\an/\bn), & \text{slow family}.
 \end{cases}
\end{equation}

\emph{Step 1: the studentised first-order field is globally bounded.}
By Cauchy--Schwarz in the $J^{-1}$ inner product,
$|d(\tau)^\top H|\le s(\tau)\,\norm{J^{1/2}H}$ for every $\tau$, including
derivative zeros (where both sides vanish). Consequently, for any positive
weight $w(\tau)\ge s(\tau)$,
\begin{equation}\label{eq:stb-cs}
 \sup_\tau\frac{|d(\tau)^\top H_{N,n}|}{w(\tau)}
 \le\norm{J^{1/2}H_{N,n}}=O_{\Pp}(1).
\end{equation}
The same bound holds for the multiplier draw $H^*$ conditionally.

\emph{Step 2: the floor absorbs the remainder.}
Since $\widehat s_\kappa\ge\kappa_{N,n}\norm{\widehat s}_\infty$ and
$\norm{\widehat s}_\infty\overset{\Pp}{\to}\norm s_\infty>0$,
\eqref{eq:stb-rem} gives
\[
 \sup_\tau\frac{|R_{N,n}(\tau)|}{\widehat s_\kappa(\tau)}
 \le\frac{\sup_\tau|R_{N,n}|}{\kappa_{N,n}\norm{\widehat s}_\infty}
 =\begin{cases}
 O_{\Pp}\big((\bn\kappa_{N,n})^{-1}\big)\to0,
 &\text{fast, using }\bn\kappa_{N,n}\to\infty,\\[2pt]
 O_{\Pp}\big((\an\kappa_{N,n})^{-1}\big)
 +O_{\Pp}\big(\tfrac{\an}{\bn\kappa_{N,n}}\big)\to0,
 &\text{slow, using both slow-family conditions}.
 \end{cases}
\]

\emph{Step 3: uniform ratio consistency of the studentiser.}
On $\{s\ge\varepsilon\}$, $\widehat s/s\to1$ uniformly in probability (from
$\widehat J\to J$, Lemma~\ref{lem:Jhat}, and uniform consistency of the
plug-in derivative surfaces), and the floor is asymptotically inactive there
because $\kappa_{N,n}\to0$; hence
$\sup_{\{s\ge\varepsilon\}}|\widehat s_\kappa/s_\kappa-1|\to0$ in
probability, where $s_\kappa:=\max\{s,\kappa_{N,n}\norm s_\infty\}$. On
$\{s<\varepsilon\}$, both $\widehat s_\kappa$ and $s_\kappa$ lie in
$[\tfrac12\kappa_{N,n}\norm s_\infty,\;\varepsilon+o_{\Pp}(1)]$, and the
uniform ratio bound
$\widehat s_\kappa/s_\kappa\in[1-\eta_\varepsilon,1+\eta_\varepsilon]$ holds
wp$\to1$ with $\eta_\varepsilon$ controlled by the same plug-in consistency.

\emph{Step 4: quantile transfer.}
Combining Steps 1--3, the coverage statistic and its bootstrap counterpart
differ by $o_{\Pp}(1)$:
\[
 \sup_\tau\frac{r|\widehat T(\tau)-T_0(\tau)|}{\widehat s_\kappa(\tau)}
 =\sup_\tau\frac{|d(\tau)^\top H_{N,n}|}{s_\kappa(\tau)}+o_{\Pp}(1),
 \qquad
 M^*_{N,n}
 =\sup_\tau\frac{|d(\tau)^\top H^*_{N,n}|}{s_\kappa(\tau)}
 +o_{\Pp^*}(1),
\]
where the first equality uses Steps 2--3 and the Lipschitz property of
$x\mapsto\sup_\tau|x(\tau)|/s_\kappa(\tau)$ under multiplicative
perturbation of the denominator together with the global bound
\eqref{eq:stb-cs} (with $w=s_\kappa\ge$ each of $s$,
$\kappa\norm s_\infty$).

\emph{Step 5: identification of the limiting functional.}
For $\kappa>0$ define $V_\kappa,V:\R^2\to[0,\infty)$ by
\[
 V_\kappa(H):=\sup_{\tau\in\cT}\frac{|d(\tau)^\top H|}{s_\kappa(\tau)},
 \qquad
 V(H):=\sup_{\tau:\,s(\tau)>0}\frac{|d(\tau)^\top H|}{s(\tau)},
\]
with the convention $V(H):=0$ if $s\equiv0$ (excluded by
$\norm s_\infty>0$). Three elementary facts close the argument.
\emph{(a) Zeros carry no signal:} since $J$ is positive definite and
$s(\tau)^2=d(\tau)^\top J^{-1}d(\tau)$, $s(\tau)=0$ iff $d(\tau)=0$; at such
$\tau$ the ratio $|d^\top H|/s_\kappa=0$ for every $\kappa>0$. Hence
$V_\kappa$ is unaffected by the zero set.
\emph{(b) Monotone pointwise convergence:} $\kappa\mapsto s_\kappa$ is
non-decreasing, so $\kappa\downarrow0$ makes $V_\kappa$ non-decreasing;
for fixed $\tau$ with $s(\tau)>0$,
$|d^\top H|/s_\kappa(\tau)\uparrow|d^\top H|/s(\tau)$. Suprema of
pointwise non-decreasing families interchange, so
$V_{\kappa}(H)\uparrow V(H)$ for every $H$ as $\kappa\downarrow0$.
\emph{(c) Uniform Lipschitz:} by \eqref{eq:stb-cs},
$|V_\kappa(H)-V_\kappa(H')|\le\norm{J^{1/2}(H-H')}$ for every $\kappa$, and
the same for $V$. By (b)--(c) and Dini-type upgrading (an equi-Lipschitz,
pointwise-monotone family converging to a Lipschitz limit converges
uniformly on compacts), $V_{\kappa_{N,n}}\to V$ uniformly on compact subsets
of $\R^2$.

Since $H_{N,n}\Rightarrow H\sim\mathcal N(0,J^{-1})$ and, conditionally in
probability, $H^*_{N,n}\Rightarrow\mathcal N(0,J^{-1})$
(Theorem~\ref{thm:parameter-bootstrap}), the extended continuous mapping
theorem \citep[Theorem~1.11.1]{vdVW1996} --- its hypotheses hold: the maps
$V_{\kappa_{N,n}}$ are deterministic, converge uniformly on compacts to the
continuous map $V$, and the arguments converge weakly --- gives
\[
 V_{\kappa_{N,n}}(H_{N,n})\Rightarrow V(H),
 \qquad
 V_{\kappa_{N,n}}(H^*_{N,n})\Rightarrow V(H)
 \ \text{conditionally in probability}.
\]
The law of $V(H)$ is continuous at its $(1-\alpha)$ quantile by assumption,
so the conditional quantiles of the bootstrap statistic converge in
probability to $q_{1-\alpha}(V(H))$, and with Step 4,
\[
 \Pp\big(T_0(\tau)\in\cC^{\mathrm{st}}_{N,n}(\tau)\ \forall\tau\big)
 =\Pp\big(V_{\kappa_{N,n}}(H_{N,n})\le q^*_{1-\alpha}+o_{\Pp}(1)\big)
 \longrightarrow1-\alpha .
\]
\end{proof}
