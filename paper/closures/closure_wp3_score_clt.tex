\begin{proof}
Throughout, expectations are under $P_{\vzero}^{N,n}$ and $C$ denotes a finite
constant depending only on $\Theta$, $T$, and the constant in
$\lambda_N\Delta_n\le x_0$; it may change between displays. Write
$\rho_k=\rho_{k,n}(\theta_0)$, $v_k=v_{k,n}(\vzero)$, $g_k=g_{k,n}(\theta_0)$,
$s_k^2=\sigma_0^2q_k^2/(2\theta_0\lambda_k)$ (the stationary modal variance),
and let
\[
 \varepsilon_{k,j}
 =\frac{Y_{k,j+1}-\rho_kY_{k,j}}{\sqrt{v_k}}
\]
be the standardised innovations, which are i.i.d.\ $\mathcal N(0,1)$ and
independent of $\mathcal F_{k,j}$.

\emph{Step 1: martingale-difference array.}
By \eqref{eq:score-theta}--\eqref{eq:score-sigma}, the per-transition scores at
$\vzero$ are
\begin{equation}\label{eq:clt-scores}
 s_{\theta,k,j}
 =-\frac{\lambda_k\Delta_n\rho_k\,Y_{k,j}}{\sqrt{v_k}}\,\varepsilon_{k,j}
 +\frac{g_k}{2}\,(\varepsilon_{k,j}^2-1),
 \qquad
 s_{\sigma,k,j}
 =\frac{1}{\sigma_0}(\varepsilon_{k,j}^2-1).
\end{equation}
Since $Y_{k,j}$ is $\mathcal F_{k,j}$-measurable and
$\E[\varepsilon_{k,j}\mid\mathcal F_{k,j}]=\E[\varepsilon_{k,j}^2-1\mid\mathcal
F_{k,j}]=0$, the array
$\xi_{k,j}:=\Dn(s_{\theta,k,j},s_{\sigma,k,j})^\top$, ordered
lexicographically in $(k,j)$, is a square-integrable martingale-difference
array with respect to the filtration generated by all modes up to the current
cell (modes are independent, so enlarging the per-mode filtration preserves
the martingale property).

\emph{Step 2: predictable quadratic variation.}
Using $\E[\varepsilon^2]=1$, $\E[\varepsilon^3]=0$, $\E[(\varepsilon^2-1)^2]=2$,
the summed conditional covariance matrix
$V_{N,n}:=\sum_{k,j}\E[\xi_{k,j}\xi_{k,j}^\top\mid\mathcal F_{k,j}]$ has entries
\begin{align}
 V^{\theta\theta}_{N,n}
 &=\frac1{\LamN}\sum_{k\le N}\sum_{j<n}
 \left\{\frac{\lambda_k^2\Delta_n^2\rho_k^2Y_{k,j}^2}{v_k}
 +\frac{g_k^2}{2}\right\},
 \qquad
 V^{\sigma\sigma}_{N,n}
 =\frac{2}{\sigma_0^2}\quad\text{(deterministic)},
 \label{eq:clt-qv}\\
 V^{\theta\sigma}_{N,n}
 &=\frac{1}{\sqrt{\LamN Nn}}\,\frac{n}{\sigma_0}\sum_{k\le N}g_k
 \quad\text{(deterministic)} .\nonumber
\end{align}
The cross term satisfies, by \eqref{eq:cross-exp} and \eqref{eq:cross-bound},
$|V^{\theta\sigma}_{N,n}|\le C\sqrt{\LamN/(Nn)}\le
C\sqrt{\lambda_N\Delta_n/T}\to0$. The $g^2$ contribution to
$V^{\theta\theta}$ is $o(1)$ by \eqref{eq:variance-theta-small}. For the
random part $W_{N}:=\LamN^{-1}\sum_{k,j}\lambda_k^2\Delta_n^2\rho_k^2Y_{k,j}^2/v_k$,
Proposition~\ref{prop:exact-info} (closed form \eqref{eq:M-closed}) and the
Riemann-sum argument of Theorem~\ref{thm:info-limit} give
$\E W_N\to T/(2\theta_0)$. For its concentration, decompose
$Y_{k,j}=\rho_k^{\,j}u_{0,k}+Z_{k,j}$ with $(Z_{k,j})_j$ centred Gaussian,
$\Cov(Z_{k,i},Z_{k,j})\le s_k^2\rho_k^{|i-j|}$. Gaussian second-moment algebra
gives $\Cov(Y_{k,i}^2,Y_{k,j}^2)\le
2s_k^4\rho_k^{2|i-j|}+4u_{0,k}^2s_k^2\rho_k^{|i-j|}$, and since
$\sum_{i,j<n}\rho_k^{|i-j|}\le 2n/(1-\rho_k)\le Cn/(\theta_0\lambda_k\Delta_n)$
(using $1-e^{-x}\ge cx$ on $[0,x_0]$),
\[
 \Var\Big(\sum_{j<n}Y_{k,j}^2\Big)
 \le C\Big(\frac{s_k^4\,n}{\lambda_k\Delta_n}
 +\frac{u_{0,k}^2s_k^2\,n}{\lambda_k\Delta_n}\Big).
\]
Multiplying by the squared coefficient
$(\lambda_k^2\Delta_n^2\rho_k^2/v_k)^2\le C\lambda_k^4\Delta_n^2/(\sigma_0^4q_k^4)$
(since $v_k\ge c\sigma_0^2q_k^2\Delta_n$ by \eqref{eq:h-exp}) and using
$s_k^2=\sigma_0^2q_k^2/(2\theta_0\lambda_k)$,
\[
 \Var\Big(\text{mode-}k\text{ term of }\LamN W_N\Big)
 \le C\Big(\lambda_kT
 +\lambda_kT\cdot\frac{\lambda_ku_{0,k}^2}{q_k^2}\Big).
\]
By independence across modes,
\[
 \Var(W_N)
 \le\frac{C T}{\LamN^2}\Big(\LamN+\lambda_N\sum_{k\le N}
 \frac{\lambda_ku_{0,k}^2}{q_k^2}\Big)
 =\frac{CT}{\LamN}
 +CT\,\frac{\lambda_N}{\LamN}\cdot\frac{R_N(u_0,Q)}{\LamN}
 \longrightarrow0
\]
by Assumption~\ref{ass:regime}(i),(iii). Hence
$V_{N,n}\overset{\Pp}{\to}J(\vzero)$.

\emph{Step 3: conditional Lyapunov condition.}
It suffices that $\sum_{k,j}\E\norm{\xi_{k,j}}^4\to0$. Using
$(a+b)^4\le8(a^4+b^4)$, $\E\varepsilon^8<\infty$, $\E Y_{k,j}^4\le
C(s_k^4+u_{0,k}^4\rho_k^{4j})$, and $v_k\ge c\sigma_0^2q_k^2\Delta_n$:
\begin{align*}
 \sum_{k,j}\E\,\xi_{\theta,k,j}^4
 &\le\frac{C}{\LamN^2}\sum_{k\le N}
 \Big\{n\,\frac{\lambda_k^4\Delta_n^4s_k^4}{v_k^2}
 +\frac{\lambda_k^4\Delta_n^4u_{0,k}^4}{v_k^2}\sum_{j<n}\rho_k^{4j}
 +n\,g_k^4\Big\}\\
 &\le\frac{C}{\LamN^2}\Big\{T\Delta_n\sum_k\lambda_k^2
 +\lambda_N\Delta_n\Big(\sum_k\frac{\lambda_ku_{0,k}^2}{q_k^2}\Big)^2
 +T\Delta_n^3\sum_k\lambda_k^4\Big\}\\
 &\le C\Big\{\frac{T\lambda_N\Delta_n}{\LamN}
 +\lambda_N\Delta_n\Big(\frac{R_N}{\LamN}\Big)^2
 +\frac{T(\lambda_N\Delta_n)^3}{\LamN}\Big\}
 \longrightarrow0,
\end{align*}
where the middle term used
$\sum_k\lambda_k^2u_{0,k}^4/q_k^4\le(\sum_k\lambda_ku_{0,k}^2/q_k^2)^2$ and
$\sum_j\rho_k^{4j}\le C/(\lambda_k\Delta_n)$. Similarly
$\sum_{k,j}\E\,\xi_{\sigma,k,j}^4\le C(Nn)^{-2}\cdot Nn=C/(Nn)\to0$, and mixed
fourth moments are controlled by Cauchy--Schwarz. The martingale central limit theorem applies in the form of
\citet[Corollary~3.1]{HallHeyde1980}; its three conditions are verified
above: (i) $(\xi_{k,j})$ is a square-integrable martingale-difference array
with respect to the nested per-$(N,n)$ filtrations (Step~1); (ii) the summed
conditional covariance converges in probability to the \emph{deterministic}
matrix $J(\vzero)$ (Step~2), so the nesting condition needed only for
stable convergence with random limits is not required; (iii) the
conditional Lindeberg condition holds, being implied by the Lyapunov
fourth-moment bound of Step~3. (For vector arrays, apply the corollary to
$a^\top\xi_{k,j}$ for each $a\in\R^2$ and use Cram\'er--Wold.) This yields
$\Delta_{N,n}\Rightarrow\mathcal N_2(0,J(\vzero))$, which is
\eqref{eq:score-clt}.

\emph{Step 4: observed information.}
Direct differentiation of \eqref{eq:score-theta}--\eqref{eq:score-sigma} shows
that each entry of $-\nabla^2_\vartheta\elln(\vzero)$ equals the corresponding
entry of the summed conditional information plus a centred term of the form
$\sum_{k,j}\{a_{k,n}Y_{k,j}\varepsilon_{k,j}
+b_{k,n}(\varepsilon_{k,j}^2-1)\}$ with deterministic coefficients satisfying
$|a_{k,n}|\le C\lambda_k^2\Delta_n^2/\sqrt{v_k}$ and
$|b_{k,n}|\le C(\lambda_k\Delta_n)^2+C\lambda_k\Delta_n/\sigma_0$
(the largest coefficients arise in the $\theta\theta$ entry; the
$\theta\sigma$ and $\sigma\sigma$ entries carry one factor
$\sigma_0^{-1}$ or $\sigma_0^{-2}$ and at most one factor $g_k$). These are
martingale sums, and after the $\Dn\cdot\Dn$ normalisation their second
moments are bounded by the same quantities as in Steps 2--3, hence vanish.
Combining with Step 2 gives \eqref{eq:observed-info}.
\end{proof}
